using UnityEngine;
using UnityEngine.InputSystem;

public class DEBUG_GlowTester : MonoBehaviour
{
    [SerializeField] private GameObject player;
    [SerializeField] private float testDuration = 5f;
    [SerializeField] private Color testColour = Color.green;

    private void Update()
    {
        if (Keyboard.current != null && Keyboard.current.gKey.wasPressedThisFrame)
        {
            DragonGlowEffect.ApplyTo(player, testDuration, testColour);
            Debug.Log("[DEBUG] Triggered glow effect");
        }
    }
}
